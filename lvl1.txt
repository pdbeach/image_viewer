def convert(self):
    try:
        points = []
        for xEdit, yEdit, zEdit in self.pointInputs:
            x = float(xEdit.text())
            y = float(yEdit.text())
            z = float(zEdit.text())
            points.append(np.array([x, y, z]))
        
        # Use first point as starting point
        startPoint = points[0]
        
        # Find the two edges from the starting point
        edge1 = points[1] - points[0]
        edge2 = points[3] - points[0]
        
        # Calculate normal (direction) as cross product of edges
        normal = np.cross(edge2, edge1)  # Swapped order to flip direction
        normalMag = np.linalg.norm(normal)
        if normalMag > 0:
            normal = normal / normalMag
        
        # Calculate sizes
        size1 = np.linalg.norm(edge1)
        size2 = np.linalg.norm(edge2)
        
        # Determine which size corresponds to which axis
        edge1Normalized = edge1 / size1 if size1 > 0 else edge1
        edge2Normalized = edge2 / size2 if size2 > 0 else edge2
        
        # Set outputs
        self.startX.setText(f"{startPoint[0]:.6g}")
        self.startY.setText(f"{startPoint[1]:.6g}")
        self.startZ.setText(f"{startPoint[2]:.6g}")
        
        self.dx.setText(f"{normal[0]:.6g}")
        self.dy.setText(f"{normal[1]:.6g}")
        self.dz.setText(f"{normal[2]:.6g}")
        
        # Size X is along edge1, Size Y is along edge2, Size Z is thickness (minimal)
        self.sizeX.setText(f"{size1:.6g}")
        self.sizeY.setText(f"{size2:.6g}")
        self.sizeZ.setText("0.001")
        
    except ValueError:
        QMessageBox.warning(self, "Input Error", "Please enter valid numbers for all coordinates.")
    except Exception as e:
        QMessageBox.warning(self, "Error", str(e))
