def computeRectangleParams(points):
p0, p1, p2, p3 = points

```
# Find adjacent points to p0 (the two closest points)
distances = [(np.linalg.norm(p - p0), i, p) for i, p in enumerate([p1, p2, p3])]
distances.sort(key=lambda x: x[0])

# Two closest are adjacent, furthest is diagonal
adj1 = distances[0][2]
adj2 = distances[1][2]

# Edge vectors from starting point
edge1 = adj1 - p0
edge2 = adj2 - p0

# Normal vector (direction perpendicular to the plane)
normal = np.cross(edge1, edge2)
normalLen = np.linalg.norm(normal)
if normalLen > 1e-10:
    normal = normal / normalLen
else:
    normal = np.array([0, 0, 1])

# Sizes are the lengths of the edges
size1 = np.linalg.norm(edge1)
size2 = np.linalg.norm(edge2)

return {
    'start': p0,
    'direction': normal,
    'sizeX': size1,
    'sizeY': size2,
    'sizeZ': 0.001,
    'edge1': edge1,
    'edge2': edge2
}
```

def convert(self):
    try:
        points = []
        for xEdit, yEdit, zEdit in self.pointInputs:
            x = float(xEdit.text())
            y = float(yEdit.text())
            z = float(zEdit.text())
            points.append(np.array([x, y, z]))
        
        result = computeRectangleParams(points)
        
        self.startX.setText(f"{result['start'][0]:.6g}")
        self.startY.setText(f"{result['start'][1]:.6g}")
        self.startZ.setText(f"{result['start'][2]:.6g}")
        
        self.dirX.setText(f"{result['direction'][0]:.6g}")
        self.dirY.setText(f"{result['direction'][1]:.6g}")
        self.dirZ.setText(f"{result['direction'][2]:.6g}")
        
        self.sizeX.setText(f"{result['sizeX']:.6g}")
        self.sizeY.setText(f"{result['sizeY']:.6g}")
        self.sizeZ.setText(f"{result['sizeZ']:.6g}")
        
        # Verify
        verifyText = self.verifyResult(points, result)
        self.verifyText.setText(verifyText)
        
    except ValueError as e:
        self.verifyText.setText(f"Error: Please enter valid numbers for all coordinates.\n{e}")

def verifyResult(self, inputPoints, result):
    start = result['start']
    edge1 = result['edge1']
    edge2 = result['edge2']
    
    # Reconstruct the 4 corners
    reconstructed = [
        start,
        start + edge1,
        start + edge2,
        start + edge1 + edge2
    ]
    
    text = "Reconstructed corners from CAD parameters:\n"
    for i, p in enumerate(reconstructed):
        text += f"  Corner {i+1}: ({p[0]:.6g}, {p[1]:.6g}, {p[2]:.6g})\n"
    
    text += "\nOriginal input points:\n"
    for i, p in enumerate(inputPoints):
        text += f"  Point {i+1}: ({p[0]:.6g}, {p[1]:.6g}, {p[2]:.6g})\n"
    
    # Check if all input points are accounted for
    matched = []
    for ip in inputPoints:
        for j, rp in enumerate(reconstructed):
            if np.allclose(ip, rp, atol=1e-6):
                matched.append(True)
                break
        else:
            matched.append(False)
    
    if all(matched):
        text += "\n✓ VERIFICATION PASSED: All input points match reconstructed corners."
    else:
        text += "\n✗ VERIFICATION FAILED: Some points don't match."
    
    return text
